38/// Errors that can occur in the network
39#[derive(Debug, Error)]
40pub enum NetworkError {
41    /// Multiple errors. Allows us to roll up multiple errors into one.
42    #[error("Multiple errors: {0:?}")]
43    Multiple(Vec<NetworkError>),
44
45    /// A configuration error
46    #[error("Configuration error: {0}")]
47    ConfigError(String),
48
49    /// An error occurred while sending a message
50    #[error("Failed to send message: {0}")]
51    MessageSendError(String),
52
53    /// An error occurred while receiving a message
54    #[error("Failed to receive message: {0}")]
55    MessageReceiveError(String),
56
57    /// The feature is unimplemented
58    #[error("Unimplemented")]
59    Unimplemented,
60
61    /// An error occurred while attempting to listen
62    #[error("Listen error: {0}")]
63    ListenError(String),
64
65    /// Failed to send over a channel
66    #[error("Channel send error: {0}")]
67    ChannelSendError(String),
68
69    /// Failed to receive over a channel
70    #[error("Channel receive error: {0}")]
71    ChannelReceiveError(String),
72
73    /// The network has been shut down and can no longer be used
74    #[error("Network has been shut down")]
75    ShutDown,
76
77    /// Failed to serialize
78    #[error("Failed to serialize: {0}")]
79    FailedToSerialize(String),
80
81    /// Failed to deserialize
82    #[error("Failed to deserialize: {0}")]
83    FailedToDeserialize(String),
84
85    /// Timed out performing an operation
86    #[error("Timeout: {0}")]
87    Timeout(String),
88
89    /// The network request had been cancelled before it could be fulfilled
90    #[error("The request was cancelled before it could be fulfilled")]
91    RequestCancelled,
92
93    /// This node does not have any peers yet
94    #[error("This node does not have any peers yet")]
95    NoPeersYet,
96
97    /// Failed to look up a node on the network
98    #[error("Node lookup failed: {0}")]
99    LookupError(String),
100}

111/// A request for some data that the consensus layer is asking for.
112#[derive(Serialize, Deserialize, Clone, Debug, PartialEq, Eq, Hash)]
113#[serde(bound(deserialize = ""))]
114pub struct DataRequest<TYPES: NodeType> {
115    /// Request
116    pub request: RequestKind<TYPES>,
117    /// View this message is for
118    pub view: ViewNumber,
119    /// signature of the Sha256 hash of the data so outsiders can't use know
120    /// public keys with stake.
121    pub signature: <TYPES::SignatureKey as SignatureKey>::PureAssembledSignatureType,
122}

124/// Underlying data request
125#[derive(Serialize, Deserialize, Clone, PartialEq, Eq, Hash)]
126pub enum RequestKind<TYPES: NodeType> {
127    /// Request VID data by our key and the VID commitment
128    Vid(ViewNumber, TYPES::SignatureKey),
129    /// Request a DA proposal for a certain view
130    DaProposal(ViewNumber),
131    /// Request for quorum proposal for a view
132    Proposal(ViewNumber),
133}
134
135impl<TYPES: NodeType> std::fmt::Debug for RequestKind<TYPES> {
136    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
137        match self {
138            RequestKind::Vid(view, key) => write!(f, "Vid({view:?}, {key})"),
139            RequestKind::DaProposal(view) => write!(f, "DaProposal({view:?})"),
140            RequestKind::Proposal(view) => write!(f, "Proposal({view:?})"),
141        }
142    }
143}

171/// represents a networking implmentration
172/// exposes low level API for interacting with a network
173/// intended to be implemented for a message transport,
174/// and memory network
175#[async_trait]
176pub trait ConnectedNetwork<K: SignatureKey + 'static>: Clone + Send + Sync + 'static {
177    /// Pauses the underlying network
178    fn pause(&self);
179
180    /// Resumes the underlying network
181    fn resume(&self);
182
183    /// Blocks until the network is successfully initialized
184    async fn wait_for_ready(&self);
185
186    /// Blocks until the network is shut down
187    fn shut_down<'a, 'b>(&'a self) -> BoxSyncFuture<'b, ()>
188    where
189        'a: 'b,
190        Self: 'b;
191
192    /// broadcast message to some subset of nodes
193    /// blocking
194    async fn broadcast_message(
195        &self,
196        view: ViewNumber,
197        message: Vec<u8>,
198        topic: Topic,
199        broadcast_delay: BroadcastDelay,
200    ) -> Result<(), NetworkError>;
201
202    /// broadcast a message only to a DA committee
203    /// blocking
204    async fn da_broadcast_message(
205        &self,
206        view: ViewNumber,
207        message: Vec<u8>,
208        recipients: Vec<K>,
209        broadcast_delay: BroadcastDelay,
210    ) -> Result<(), NetworkError>;
211
212    /// send messages with vid shares to its recipients
213    /// blocking
214    async fn vid_broadcast_message(
215        &self,
216        messages: HashMap<K, (ViewNumber, Vec<u8>)>,
217    ) -> Result<(), NetworkError> {
218        let future_results = messages
219            .into_iter()
220            .map(|(recipient_key, (v, m))| self.direct_message(v, m, recipient_key));
221        let results = join_all(future_results).await;
222
223        let errors: Vec<_> = results.into_iter().filter_map(|r| r.err()).collect();
224
225        if errors.is_empty() {
226            Ok(())
227        } else {
228            Err(NetworkError::Multiple(errors))
229        }
230    }
231
232    /// Sends a direct message to a specific node
233    /// blocking
234    async fn direct_message(
235        &self,
236        view: ViewNumber,
237        message: Vec<u8>,
238        recipient: K,
239    ) -> Result<(), NetworkError>;
240
241    /// Receive one or many messages from the underlying network.
242    ///
243    /// # Errors
244    /// If there is a network-related failure.
245    async fn recv_message(&self) -> Result<Vec<u8>, NetworkError>;
246
247    /// queues lookup of a node
248    ///
249    /// # Errors
250    /// Does not error.
251    fn queue_node_lookup(
252        &self,
253        _: ViewNumber,
254        _: K,
255    ) -> Result<(), TrySendError<Option<(ViewNumber, K)>>> {
256        Ok(())
257    }
258
259    /// Update view can be used for any reason, but mostly it's for canceling tasks,
260    /// and looking up the address of the leader of a future view.
261    async fn update_view<TYPES>(
262        &self,
263        _: ViewNumber,
264        _: Option<EpochNumber>,
265        _: EpochMembershipCoordinator<TYPES>,
266    ) where
267        TYPES: NodeType<SignatureKey = K>,
268    {
269    }
270
271    /// Is primary network down? Makes sense only for combined network
272    fn is_primary_down(&self) -> bool {
273        false
274    }
275}

287/// Used when broadcasting messages
288#[derive(Clone, Debug, PartialEq, Eq, Hash, Copy)]
289pub enum Topic {
290    /// The `Global` topic goes out to all nodes
291    Global,
292    /// The `Da` topic goes out to only the DA committee
293    Da,
294}
295
296/// Topics travel as strings, so we need to convert our enum to a string
297impl Display for Topic {
298    fn fmt(&self, f: &mut std::fmt::Formatter<'_>) -> std::fmt::Result {
299        match self {
300            Topic::Global => write!(f, "global"),
301            Topic::Da => write!(f, "DA"),
302        }
303    }
