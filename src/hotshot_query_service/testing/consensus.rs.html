28#[async_trait]
29pub trait DataSourceLifeCycle: Clone + Send + Sync + Sized + 'static {
30    /// Backing storage for the data source.
31    ///
32    /// This can be used to connect to data sources to the same underlying data. It must be kept
33    /// alive as long as the related data sources are open.
34    type Storage: Send + Sync;
35
36    /// Type parameter for builder.
37    type S;
38
39    /// Type parameter for builder.
40    type P;
41
42    async fn create(node_id: usize) -> Self::Storage;
43    async fn build(
44        storage: &Self::Storage,
45        opt: impl Send
46        + FnOnce(Builder<MockTypes, Self::S, Self::P>) -> Builder<MockTypes, Self::S, Self::P>,
47    ) -> Self;
48    async fn reset(storage: &Self::Storage) -> Self;
49    async fn handle_event(&self, event: &CoordinatorEvent<MockTypes>);
50
51    async fn connect(storage: &Self::Storage) -> Self {
52        Self::build(storage, |builder| builder).await
53    }
54
55    async fn leaf_only_ds(storage: &Self::Storage) -> Self {
56        Self::build(storage, |builder| builder.leaf_only()).await
57    }
58}
