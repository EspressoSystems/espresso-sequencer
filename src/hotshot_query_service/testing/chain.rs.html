47/// One decided block with everything a data source stores for it.
48#[derive(Clone, Debug)]
49pub struct MockBlock {
50    pub leaf: LeafQueryData<MockTypes>,
51    pub block: BlockQueryData<MockTypes>,
52    pub vid_common: VidCommonQueryData<MockTypes>,
53    /// One share per storage node; genesis has `GENESIS_VID_NUM_STORAGE_NODES` shares.
54    pub vid_shares: Vec<VidShare>,
55}
56
57impl MockBlock {
58    pub fn height(&self) -> u64 {
59        self.leaf.leaf().block_header().block_number
60    }
61
62    /// Everything node `node` would append for this block on decide.
63    pub fn block_info(&self, node: usize) -> BlockInfo<MockTypes> {
64        BlockInfo::new(
65            self.leaf.clone(),
66            Some(self.block.clone()),
67            Some(self.vid_common.clone()),
68            self.vid_shares.get(node).cloned(),
69        )
70    }
71}

73/// A chain of decided blocks at [`NEW_PROTOCOL_VERSION`], starting from genesis.
74///
75/// Each leaf's `justify_qc` certifies its parent, and each [`LeafQueryData`] carries a QC
76/// certifying the leaf itself, so the chain is consistent the way a real decide is. QC signatures
77/// are not real; the query service does not check them.
78#[derive(Clone, Debug)]
79pub struct MockChain {
80    blocks: Vec<MockBlock>,
81    num_nodes: usize,
82    epoch_height: u64,
83}
84
85impl MockChain {
86    pub async fn new(num_nodes: usize, epoch_height: u64) -> Self {
87        Self {
88            blocks: vec![genesis(epoch_height).await],
89            num_nodes,
90            epoch_height,
91        }
92    }

94    pub fn blocks(&self) -> &[MockBlock] {
95        &self.blocks
96    }
97
98    pub fn tip(&self) -> &MockBlock {
99        self.blocks.last().expect("chain always has genesis")
100    }

102    /// Extend the chain by one block containing `txs`, one second after the tip.
103    pub async fn push(
104        &mut self,
105        txs: impl IntoIterator<Item = MockTransaction> + Send,
106    ) -> &MockBlock {
107        let timestamp = self.tip().leaf.leaf().block_header().timestamp + 1;
108        self.push_at(txs, timestamp).await
109    }
110
111    /// Extend the chain by one block containing `txs`, with header timestamp `timestamp` seconds.
112    pub async fn push_at(
113        &mut self,
114        txs: impl IntoIterator<Item = MockTransaction> + Send,
115        timestamp: u64,
116    ) -> &MockBlock {
117        let (payload, metadata) = <MockPayload as BlockPayload<MockTypes>>::from_transactions(
118            txs,
119            &Default::default(),
120            &Default::default(),
121        )
122        .await
123        .unwrap();
124        let (payload_commitment, vid_common, vid_shares) =
125            disperse(&payload, &metadata, self.num_nodes);
126
127        let parent = self.tip().leaf.leaf();
128        let height = parent.block_header().block_number + 1;
129        let header = MockHeader {
130            block_number: height,
131            payload_commitment,
132            builder_commitment: <MockPayload as BlockPayload<MockTypes>>::builder_commitment(
133                &payload, &metadata,
134            ),
135            metadata,
136            timestamp,
137            timestamp_millis: timestamp * 1000,
138            random: 0,
139            version: NEW_PROTOCOL_VERSION,
140        };
141        let leaf = Leaf2::from_quorum_proposal(&QuorumProposalWrapper {
142            proposal: QuorumProposal2 {
143                block_header: header.clone(),
144                view_number: parent.view_number() + 1,
145                epoch: Some(self.epoch(height)),
146                justify_qc: qc_for(parent, self.epoch_height),
147                next_epoch_justify_qc: None,
148                upgrade_certificate: None,
149                view_change_evidence: None,
150                next_drb_result: None,
151                state_cert: None,
152            },
153        });
154        let qc = qc_for(&leaf, self.epoch_height);
155
156        self.blocks.push(MockBlock {
157            leaf: LeafQueryData::new(leaf, qc).unwrap(),
158            block: BlockQueryData::new(header.clone(), payload),
159            vid_common: VidCommonQueryData::new(header, vid_common),
160            vid_shares,
161        });
162        self.tip()
163    }

165    /// Extend the chain by `n` empty blocks.
166    pub async fn push_empty(&mut self, n: usize) {
167        for _ in 0..n {
168            self.push([]).await;
169        }
170    }

172    /// The event consensus sends node `node` when the blocks at `heights` are decided together.
173    ///
174    /// As the coordinator sends it, leaves are newest first and carry their payloads, `cert1`
175    /// certifies the newest leaf and `cert2` finalizes it. Genesis is the exception: consensus
176    /// neither disperses it nor finalizes it with a cert2, so it carries no VID share and its
177    /// decide has no cert2.
178    pub fn decide_event(
179        &self,
180        heights: RangeInclusive<usize>,
181        node: usize,
182    ) -> CoordinatorEvent<MockTypes> {
183        let blocks = &self.blocks[heights];
184        let newest = blocks.last().expect("decide at least one block");
185        let leaf_infos = blocks
186            .iter()
187            .rev()
188            .map(|block| {
189                let vid_share = (block.height() > 0).then(|| self.vid_disperse_share(block, node));
190                let mut leaf = block.leaf.leaf().clone();
191                leaf.fill_block_payload_unchecked(block.block.payload().clone());
192                LeafInfo::new(leaf, Arc::new(Default::default()), None, vid_share, None)
193            })
194            .collect();
195        CoordinatorEvent::NewDecide {
196            leaf_infos,
197            cert1: newest.leaf.qc().clone(),
198            cert2: self.cert2(newest),
199        }
200    }
201
202    /// The cert2 finalizing `block`, or `None` for genesis, which consensus decides without one.
203    pub fn cert2(&self, block: &MockBlock) -> Option<Certificate2<MockTypes>> {
204        (block.height() > 0).then(|| self.cert2_for(block.leaf.leaf()))
205    }
206
207    fn vid_disperse_share(&self, block: &MockBlock, node: usize) -> VidDisperseShare<MockTypes> {
208        let (VidCommitment::V2(payload_commitment), VidCommon::V2(common), VidShare::V2(share)) = (
209            block.leaf.payload_hash(),
210            block.vid_common.common().clone(),
211            block.vid_shares[node].clone(),
212        ) else {
213            panic!("chain disperses with AvidmGf2");
214        };
215        let epoch = Some(self.epoch(block.height()));
216        VidDisperseShare::V2(AvidmGf2DisperseShare {
217            view_number: block.leaf.leaf().view_number(),
218            epoch,
219            target_epoch: epoch,
220            payload_commitment,
221            share,
222            recipient_key: BLSPubKey::generated_from_seed_indexed([0; 32], node as u64).0,
223            common,
224        })
225    }
226
227    fn cert2_for(&self, leaf: &Leaf2<MockTypes>) -> Certificate2<MockTypes> {
228        let height = leaf.block_header().block_number;
229        let data = Vote2Data {
230            leaf_commit: leaf.commit(),
231            epoch: self.epoch(height),
232            block_number: height,
233        };
234        let commit = data.commit();
235        Certificate2::new(data, commit, leaf.view_number(), None, PhantomData)
236    }
237
238    fn epoch(&self, height: u64) -> EpochNumber {
239        EpochNumber::new(epoch_from_block_number(height, self.epoch_height))
240    }
241}

243/// Storage nodes the chain disperses VID to.
244pub const NUM_NODES: usize = 2;
245
246pub const EPOCH_HEIGHT: u64 = 10;

248/// A data source following a [`MockChain`]: each block pushed onto the chain is decided and
249/// appended to the data source with the cert2 finalizing it, as a node following consensus
250/// would.
251pub struct ChainNode<D: DataSourceLifeCycle> {
252    chain: MockChain,
253    data_source: D,
254    // Dropping the storage (a temp dir or database) would tear it down under the data source.
255    _storage: D::Storage,
256}
257
258impl<D: DataSourceLifeCycle + UpdateAvailabilityData<MockTypes>> ChainNode<D> {
259    /// A node that has decided genesis.
260    pub async fn new() -> Self {
261        let storage = D::create(0).await;
262        let data_source = D::connect(&storage).await;
263        Self::with_data_source(storage, data_source).await
264    }
265
266    /// A node with a leaf-only data source that has decided genesis.
267    pub async fn leaf_only() -> Self {
268        let storage = D::create(0).await;
269        let data_source = D::leaf_only_ds(&storage).await;
270        Self::with_data_source(storage, data_source).await
271    }
272
273    pub fn data_source(&self) -> D {
274        self.data_source.clone()
275    }
276
277    pub fn chain(&self) -> &MockChain {
278        &self.chain
279    }
280
281    /// Decide one block containing `txs`, one second after the tip.
282    pub async fn push(
283        &mut self,
284        txs: impl IntoIterator<Item = MockTransaction> + Send,
285    ) -> MockBlock {
286        let block = self.chain.push(txs).await.clone();
287        self.decide(&block).await;
288        block
289    }
290
291    /// Decide one block containing `txs`, with header timestamp `timestamp` seconds.
292    pub async fn push_at(
293        &mut self,
294        txs: impl IntoIterator<Item = MockTransaction> + Send,
295        timestamp: u64,
296    ) -> MockBlock {
297        let block = self.chain.push_at(txs, timestamp).await.clone();
298        self.decide(&block).await;
299        block
300    }
301
302    /// Decide `n` empty blocks.
303    pub async fn push_empty(&mut self, n: usize) {
304        for _ in 0..n {
305            self.push([]).await;
306        }
307    }
308
309    async fn with_data_source(storage: D::Storage, data_source: D) -> Self {
310        let mut node = Self {
311            chain: MockChain::new(NUM_NODES, EPOCH_HEIGHT).await,
312            data_source,
313            _storage: storage,
314        };
315        let genesis = node.chain.tip().clone();
316        node.decide(&genesis).await;
317        node
318    }
319
320    async fn decide(&mut self, block: &MockBlock) {
321        let mut info = block.block_info(0);
322        if let Some(cert2) = self.chain.cert2(block) {
323            info = info.with_cert2(cert2);
324        }
325        self.data_source.append(info).await.unwrap();
326    }
327}

329async fn genesis(epoch_height: u64) -> MockBlock {
330    let leaf = Leaf2::<MockTypes>::genesis(
331        &Default::default(),
332        &Default::default(),
333        NEW_PROTOCOL_VERSION,
334    )
335    .await;
336    let header = leaf.block_header().clone();
337    let payload = MockPayload::genesis();
338
339    // The genesis header commits to VID over `GENESIS_VID_NUM_STORAGE_NODES`, so disperse to
340    // exactly that many nodes for the commitment to match.
341    let (payload_commitment, vid_common, vid_shares) =
342        disperse(&payload, &header.metadata, GENESIS_VID_NUM_STORAGE_NODES);
343    assert_eq!(payload_commitment, header.payload_commitment);
344
345    let qc = qc_for(&leaf, epoch_height);
346    MockBlock {
347        leaf: LeafQueryData::new(leaf, qc).unwrap(),
348        block: BlockQueryData::new(header.clone(), payload),
349        vid_common: VidCommonQueryData::new(header, vid_common),
350        vid_shares,
351    }
352}
353
354/// A QC certifying `leaf`, with the epoch its height falls in under `epoch_height`.
355fn qc_for(leaf: &Leaf2<MockTypes>, epoch_height: u64) -> QuorumCertificate2<MockTypes> {
356    let height = leaf.block_header().block_number;
357    let data = QuorumData2 {
358        leaf_commit: leaf.commit(),
359        epoch: Some(EpochNumber::new(epoch_from_block_number(
360            height,
361            epoch_height,
362        ))),
363        block_number: Some(height),
364    };
365    QuorumCertificate2::new(data, data.commit(), leaf.view_number(), None, PhantomData)
366}
367
368/// AvidmGf2 dispersal of `payload` over `num_nodes` equally weighted nodes.
369fn disperse(
370    payload: &MockPayload,
371    metadata: &<MockPayload as BlockPayload<MockTypes>>::Metadata,
372    num_nodes: usize,
373) -> (VidCommitment, VidCommon, Vec<VidShare>) {
374    let bytes = payload.encode();
375    let param = init_avidm_gf2_param(num_nodes).unwrap();
376    let ns_table = parse_ns_table(bytes.len(), &metadata.encode());
377    let (commit, common, shares) =
378        AvidmGf2Scheme::ns_disperse(&param, &vec![1; num_nodes], &bytes, ns_table).unwrap();
379    (
380        VidCommitment::V2(commit),
381        VidCommon::V2(common),
382        shares.into_iter().map(VidShare::V2).collect(),
383    )
384}

398    #[tokio::test]
399    async fn test_chain_is_linked() {
400        let mut chain = MockChain::new(3, 10).await;
401        chain.push_empty(2).await;
402        chain.push([mock_transaction(vec![1, 2, 3])]).await;
403        chain.push_at([], 100).await;
404
405        for (parent, child) in chain.blocks().iter().zip(&chain.blocks()[1..]) {
406            assert_eq!(child.height(), parent.height() + 1);
407            assert_eq!(
408                child.leaf.leaf().parent_commitment(),
409                parent.leaf.leaf().commit()
410            );
411            assert_eq!(child.vid_shares.len(), 3);
412        }
413        assert_eq!(chain.blocks()[3].block.num_transactions(), 1);
414        assert_eq!(chain.tip().leaf.header().timestamp, 100);
415    }

417    #[tokio::test]
418    async fn test_chain_round_trips_through_storage() {
419        let storage = MockSqlDataSource::create(0).await;
420        let ds = <MockSqlDataSource as DataSourceLifeCycle>::connect(&storage).await;
421
422        let mut chain = MockChain::new(2, 10).await;
423        chain.push([mock_transaction(vec![1, 2, 3])]).await;
424        chain.push_empty(1).await;
425        for block in chain.blocks() {
426            ds.append(block.block_info(1)).await.unwrap();
427        }
428
429        for block in chain.blocks() {
430            let height = block.height() as usize;
431            assert_eq!(ds.get_leaf(height).await.await, block.leaf);
432            assert_eq!(ds.get_block(height).await.await, block.block);
433            assert_eq!(ds.get_vid_common(height).await.await, block.vid_common);
434        }
435        assert_eq!(chain.blocks()[1].block.num_transactions(), 1);
436        assert_eq!(
437            ds.vid_share(1).await.unwrap(),
438            chain.blocks()[1].vid_shares[1]
439        );
440    }
