1//! Chains recorded from legacy consensus (before 0.6), replayed through a query node that runs
2//! no consensus: a follower fed the recorded blocks, on an anvil that mines the recorded L1
3//! transactions again, so the stake table, rewards and merklized state are derived by the code a
4//! live node runs. Mainnet and decaf still serve pre-0.6 data, and once legacy consensus is gone
5//! nothing else can produce it; `data/legacy-chains/README.md` describes the chains.
6//!
7//! The recordings cannot be regenerated once legacy consensus is deleted; the recorder is kept on
8//! the `legacy-chain-recorder` branch. A change to how any recorded type deserializes has to keep
9//! reading them, as the `data/v*` reference vectors do.

60/// A chain recorded from legacy consensus, as stored under `data/legacy-chains/<name>`.
61pub struct LegacyChain {
62    pub name: String,
63    pub genesis: Genesis,
64    pub network_config: NetworkConfig<SeqTypes>,
65    pub blocks: Vec<RecordedBlock>,
66    /// One per finished epoch, keyed by epoch.
67    pub state_certs: Vec<(u64, LightClientStateUpdateCertificateV2<SeqTypes>)>,
68    /// The network config registered on the stake table's V3 contract, if the recording did.
69    pub network_config_update: Option<RecordedNetworkConfigUpdate>,
70    /// Indexed by block number; block 0 is anvil's genesis and has no transactions.
71    pub l1_blocks: Vec<RecordedL1Block>,
72}

81/// An L1 block as its timestamp and signed transactions, which is all a replay needs to mine it
82/// again: contract addresses, logs and block numbers depend only on the transactions and their
83/// order. Block hashes do not survive a replay, and nothing the node runs compares them.
84#[derive(Clone, Debug, PartialEq, Eq, Serialize, Deserialize)]
85pub struct RecordedL1Block {
86    pub timestamp: u64,
87    /// In the order mined, as `eth_sendRawTransaction` takes them.
88    pub transactions: Vec<Bytes>,
89}

107impl LegacyChain {
108    pub fn load(name: &str) -> anyhow::Result<Self> {
109        let dir = chain_dir(name);
110        let read = |file: &str| {
111            fs::read_to_string(dir.join(file)).with_context(|| format!("reading {name}/{file}"))
112        };
113        let network_config_update = if dir.join(NETWORK_CONFIG_UPDATE).exists() {
114            Some(serde_json::from_str(&read(NETWORK_CONFIG_UPDATE)?)?)
115        } else {
116            None
117        };
118        Ok(Self {
119            name: name.into(),
120            genesis: Genesis::from_file(dir.join(GENESIS))?,
121            network_config: serde_json::from_str(&read(NETWORK_CONFIG)?)?,
122            blocks: serde_json::from_str(&read(BLOCKS)?)?,
123            state_certs: serde_json::from_str(&read(STATE_CERTS)?)?,
124            network_config_update,
125            l1_blocks: serde_json::from_str(&read(L1_BLOCKS)?)?,
126        })
127    }
128
129    /// The height of the last recorded block.
130    pub fn tip(&self) -> u64 {
131        self.blocks
132            .last()
133            .expect("a recorded chain has blocks")
134            .leaf
135            .height()
136    }

138    /// Starts a query node on this chain and returns once it has stored every block and derived
139    /// the merklized state up to the tip.
140    pub async fn replay(&self) -> anyhow::Result<ReplayNode> {
141        let l1 = replay_l1(&self.l1_blocks)
142            .await
143            .with_context(|| format!("replaying the {} chain's L1", self.name))?;
144
145        let storage = SqlDataSource::create_storage().await;
146        let mut db = tmp_options(&storage);
147        let persistence = db.create().await?;
148        // There are no peers to fetch the config or state certs from.
149        persistence.save_config(&self.network_config).await?;
150        for (epoch, cert) in &self.state_certs {
151            persistence.insert_state_cert(*epoch, cert.clone()).await?;
152        }
153
154        let genesis = self.genesis.clone();
155        let l1_params = L1Params {
156            urls: vec![l1.endpoint_url()],
157            options: Default::default(),
158        };
159        let source = Arc::new(ReplaySource {
160            blocks: self.blocks.clone(),
161        });
162        let port = reserve_tcp_port().context("no free port for the replayed query API")?;
163        let url: url::Url = format!("http://localhost:{port}").parse()?;
164        let params = replay_params(url.clone());
165        let handle = Options::with_port(port)
166            .query_sql(Query::default(), db)
167            .catchup(Default::default())
168            .config(Default::default())
169            .explorer(Default::default())
170            .serve(move |metrics, sink, _| {
171                async move {
172                    init_replay_node(
173                        genesis,
174                        params,
175                        metrics,
176                        persistence,
177                        l1_params,
178                        sink,
179                        source,
180                    )
181                    .await
182                }
183                .boxed()
184            })
185            .await
186            .context("replayed node should start")?;
187        let node = FollowerContext::new(handle);
188
189        let client = Client::new(url);
190        ensure!(
191            client.connect(Some(Duration::from_secs(60))).await,
192            "replayed query API did not come up"
193        );
194        wait_for_replay(&client, self.tip()).await?;
195        Ok(ReplayNode {
196            node,
197            client,
198            l1,
199            _storage: storage,
200        })
201    }
202}

217/// Starts an anvil at the recorded genesis timestamp and mines the recorded blocks one by one.
218async fn replay_l1(blocks: &[RecordedL1Block]) -> anyhow::Result<AnvilInstance> {
219    let (genesis, rest) = blocks
220        .split_first()
221        .context("the recorded L1 has a genesis block")?;
222    // As the recorder ran anvil: every block is finalized at once, and the default accounts hold
223    // enough to fund the recorded transactions.
224    let anvil = Anvil::new()
225        .args([
226            "--slots-in-an-epoch",
227            "0",
228            "--balance",
229            "1000000",
230            "--no-mining",
231        ])
232        .args(["--timestamp", &genesis.timestamp.to_string()])
233        .spawn();
234    let provider = ProviderBuilder::new().connect_http(anvil.endpoint_url());
235    let mined = l1_block(&provider, 0).await?;
236    ensure!(
237        mined.header.timestamp == genesis.timestamp,
238        "anvil's genesis block has timestamp {} instead of the recorded {}",
239        mined.header.timestamp,
240        genesis.timestamp
241    );
242    for (number, block) in (1u64..).zip(rest) {
243        mine_l1_block(&provider, number, block).await?;
244    }
245    Ok(anvil)
246}

248async fn mine_l1_block(
249    provider: &impl Provider,
250    number: u64,
251    block: &RecordedL1Block,
252) -> anyhow::Result<()> {
253    provider
254        .anvil_set_next_block_timestamp(block.timestamp)
255        .await
256        .with_context(|| format!("setting the timestamp of L1 block {number}"))?;
257    let mut sent = vec![];
258    for (i, tx) in block.transactions.iter().enumerate() {
259        // Not awaited: with mining off a pending transaction never confirms.
260        let pending = provider
261            .send_raw_transaction(tx)
262            .await
263            .with_context(|| format!("sending transaction {i} of L1 block {number}"))?;
264        sent.push(*pending.tx_hash());
265    }
266    provider.anvil_mine(Some(1), None).await?;
267    let mined = l1_block(provider, number).await?;
268    ensure!(
269        mined.header.timestamp == block.timestamp,
270        "L1 block {number} was mined at {} instead of the recorded {}",
271        mined.header.timestamp,
272        block.timestamp
273    );
274    let hashes: Vec<_> = mined.transactions.hashes().collect();
275    ensure!(
276        hashes == sent,
277        "L1 block {number} holds transactions {hashes:?} instead of the recorded {sent:?}"
278    );
279    Ok(())
280}

282async fn l1_block(provider: &impl Provider, number: u64) -> anyhow::Result<Block> {
283    provider
284        .get_block_by_number(number.into())
285        .await?
286        .with_context(|| format!("anvil has no L1 block {number}"))
287}

289/// The node is its own only peer: it serves catchup from the state it derived itself.
290fn replay_params(own_url: url::Url) -> FollowerParams {
291    FollowerParams {
292        upstreams: vec![],
293        config_peers: None,
294        catchup: CatchupParams {
295            state_peers: vec![own_url],
296            backoff: Default::default(),
297            base_timeout: Duration::from_secs(2),
298            local_timeout: Duration::from_secs(5),
299        },
300        bootstrap_epoch_catchup_timeout: Duration::from_secs(30),
301        options: FollowerOptions {
302            poll_interval: Duration::from_millis(100),
303            // Skipping ahead would leave the gap to a backfill with nothing to fetch from.
304            max_blocks_per_poll: u64::MAX,
305            light_client: Default::default(),
306            light_client_db: Default::default(),
307            light_client_genesis: None,
308        },
309    }
310}

312async fn wait_for_replay(
313    client: &Client<ClientErr, SequencerApiVersion>,
314    tip: u64,
315) -> anyhow::Result<()> {
316    wait_for_height(
317        client,
318        "node/block-height",
319        tip + 1,
320        Duration::from_secs(120),
321    )
322    .await
323    .context("storing the recorded blocks stalled")?;
324    wait_for_height(
325        client,
326        "block-state/block-height",
327        tip,
328        Duration::from_secs(300),
329    )
330    .await
331    .context("rebuilding the merklized state stalled; the state loop logs which root differs")
332}

334async fn wait_for_height(
335    client: &Client<ClientErr, SequencerApiVersion>,
336    route: &str,
337    target: u64,
338    timeout: Duration,
339) -> anyhow::Result<()> {
340    let deadline = Instant::now() + timeout;
341    let mut reached = None;
342    while Instant::now() < deadline {
343        if let Ok(height) = client.get::<u64>(route).send().await {
344            if height >= target {
345                return Ok(());
346            }
347            reached = Some(height);
348        }
349        sleep(Duration::from_millis(200)).await;
350    }
351    bail!("{route} reached {reached:?} of {target} in {timeout:?}")
352}

354/// Hands a follower the recorded blocks, as a light client hands it verified ones.
355struct ReplaySource {
356    blocks: Vec<RecordedBlock>,
357}
358
359#[async_trait]
360impl ChainSource for ReplaySource {
361    async fn block_height(&self) -> anyhow::Result<u64> {
362        Ok(self.blocks.len() as u64)
363    }
364
365    async fn fetch_range(&self, from: u64, to: u64) -> anyhow::Result<Vec<VerifiedBlock>> {
366        let blocks = self
367            .blocks
368            .get(from as usize..to as usize)
369            .with_context(|| format!("blocks {from}..{to} were not recorded"))?;
370        Ok(blocks
371            .iter()
372            .cloned()
373            .map(
374                |RecordedBlock {
375                     leaf,
376                     block,
377                     vid_common,
378                 }| VerifiedBlock {
379                    leaf,
380                    block,
381                    vid_common,
382                },
383            )
384            .collect())
385    }
386
387    /// No legacy block has one.
388    async fn fetch_cert2(
389        &self,
390        _header: &Header,
391    ) -> anyhow::Result<Option<Certificate2<SeqTypes>>> {
392        Ok(None)
393    }
