38pub(crate) async fn compute_state_update(
39    parent_state: &ValidatedState,
40    instance: &NodeState,
41    peers: &impl StateCatchup,
42    parent_leaf: &Leaf2,
43    proposed_leaf: &Leaf2,
44) -> anyhow::Result<(ValidatedState, Delta)> {
45    let header = proposed_leaf.block_header();
46
47    let mut parent_state = parent_state.clone();
48
49    // if the protocol has been upgraded, the new chain_config should be used
50    // as the base chain config for the call to `apply_header`. This mirrors the
51    // `apply_upgrade` step at the start of `apply_header`
52    //
53    // We need to do this here because this loop may need to process historical upgrades
54    // that are no longer recorded in our genesis file. but it's safe, because this loop
55    // only handles decided leaves
56    if proposed_leaf.block_header().version() > parent_leaf.block_header().version() {
57        parent_state.chain_config = proposed_leaf.block_header().chain_config()
58    }
59
60    let (state, delta, total_rewards_distributed) = parent_state
61        .apply_header(
62            instance,
63            peers,
64            parent_leaf,
65            header,
66            header.version(),
67            proposed_leaf.view_number(),
68        )
69        .await?;
70
71    // Check internal consistency.
72    ensure!(
73        state.chain_config.commit() == header.chain_config().commit(),
74        "internal error! in-memory chain config {:?} does not match header {:?}",
75        state.chain_config,
76        header.chain_config(),
77    );
78    ensure!(
79        state.block_merkle_tree.commitment() == header.block_merkle_tree_root(),
80        "internal error! in-memory block tree {} does not match header {}",
81        state.block_merkle_tree.commitment(),
82        header.block_merkle_tree_root()
83    );
84    ensure!(
85        state.fee_merkle_tree.commitment() == header.fee_merkle_tree_root(),
86        "internal error! in-memory fee tree {} does not match header {}",
87        state.fee_merkle_tree.commitment(),
88        header.fee_merkle_tree_root()
89    );
90
91    match header.reward_merkle_tree_root() {
92        Either::Left(v1_root) => {
93            ensure!(
94                state.reward_merkle_tree_v1.commitment() == v1_root,
95                "internal error! in-memory v1 reward tree {} does not match header {}",
96                state.reward_merkle_tree_v1.commitment(),
97                v1_root
98            )
99        },
100        Either::Right(v2_root) => {
101            ensure!(
102                state.reward_merkle_tree_v2.commitment() == v2_root,
103                "internal error! in-memory v2 reward tree {} does not match header {}",
104                state.reward_merkle_tree_v2.commitment(),
105                v2_root
106            )
107        },
108    }
109
110    if header.version() >= DRB_AND_HEADER_UPGRADE_VERSION {
111        let Some(actual_total) = total_rewards_distributed else {
112            bail!(
113                "internal error! total_rewards_distributed is None for version {:?}",
114                header.version()
115            );
116        };
117
118        let Some(proposed_total) = header.total_reward_distributed() else {
119            bail!(
120                "internal error! proposed header.total_reward_distributed() is None for version \
121                 {:?}",
122                header.version()
123            );
124        };
125
126        ensure!(
127            proposed_total == actual_total,
128            "Total rewards mismatch: proposed header has {proposed_total} but actual total is \
129             {actual_total}",
130        );
131    }
132
133    Ok((state, delta))
134}

136async fn store_state_update(
137    tx: &mut impl SequencerStateUpdate,
138    block_number: u64,
139    _version: Version,
140    state: &ValidatedState,
141    delta: &Delta,
142) -> anyhow::Result<()> {
143    let ValidatedState {
144        fee_merkle_tree,
145        block_merkle_tree,
146        ..
147    } = state;
148    let Delta { fees_delta, .. } = delta;
149
150    // Collect fee merkle tree proofs for batch insertion
151    let fee_proofs: Vec<_> = fees_delta
152        .iter()
153        .map(|delta| {
154            let proof = match fee_merkle_tree.universal_lookup(*delta) {
155                LookupResult::Ok(_, proof) => proof,
156                LookupResult::NotFound(proof) => proof,
157                LookupResult::NotInMemory => bail!("missing merkle path for fee account {delta}"),
158            };
159            let path = FeeAccount::to_traversal_path(delta, fee_merkle_tree.height());
160            Ok((proof, path))
161        })
162        .collect::<anyhow::Result<Vec<_>>>()?;
163
164    tracing::debug!(count = fee_proofs.len(), "inserting fee accounts in batch");
165    UpdateStateData::<SeqTypes, FeeMerkleTree, { FeeMerkleTree::ARITY }>::insert_merkle_nodes_batch(
166        tx,
167        fee_proofs,
168        block_number,
169    )
170    .await
171    .context("failed to store fee merkle nodes")?;
172
173    // Insert block merkle tree nodes
174    let (_, proof) = block_merkle_tree
175        .lookup(block_number - 1)
176        .expect_ok()
177        .context("getting blocks frontier")?;
178    let path = <u64 as ToTraversalPath<{ BlockMerkleTree::ARITY }>>::to_traversal_path(
179        &(block_number - 1),
180        block_merkle_tree.height(),
181    );
182
183    {
184        tracing::debug!("inserting blocks frontier");
185        UpdateStateData::<SeqTypes, BlockMerkleTree, { BlockMerkleTree::ARITY }>::insert_merkle_nodes(
186            tx,
187            proof,
188            path,
189            block_number,
190        )
191        .await
192        .context("failed to store block merkle nodes")?;
193    }
194
195    Ok(())
196}

206async fn update_state_storage<T>(
207    parent_state: &ValidatedState,
208    storage: &Arc<T>,
209    instance: &NodeState,
210    peers: &impl StateCatchup,
211    parent_leaf: &Leaf2,
212    proposed_leaf: &Leaf2,
213) -> anyhow::Result<ValidatedState>
214where
215    T: SequencerStateDataSource,
216    for<'a> T::Transaction<'a>: SequencerStateUpdate,
217{
218    let parent_chain_config = parent_state.chain_config;
219    let block_number = proposed_leaf.height();
220    let version = proposed_leaf.block_header().version();
221
222    let (state, delta) =
223        compute_state_update(parent_state, instance, peers, parent_leaf, proposed_leaf)
224            .await
225            .context("computing state update")?;
226
227    let has_changed_accounts = version > EPOCH_VERSION && !delta.rewards_delta.is_empty();
228    // For EPOCH_REWARD_VERSION+ we must persist the reward tree at every epoch
229    // boundary, even when no rewards were distributed. During a V4→V5 upgrade
230    // the first post upgrade epoch boundary skips rewards (the previous epoch's
231    // header is pre-V5), leaving rewards_delta empty. Without saving here, the
232    // tree would be missing from storage and catchup requests from peers or
233    // subsequent epoch reward calculations would fail.
234    //
235    // Example: V4→V5 upgrade at block 9756, epoch_height=3000.
236    // At block 12000 (first epoch boundary post upgrade), handle_epoch_rewards
237    // skips rewards because the previous epoch's boundary (block 9000) is pre-V5,
238    // so rewards_delta is empty. Without saving here, the tree is never persisted
239    // at height 12000. Later at block 15000, the epoch 4 reward calculation calls
240    // fetch_reward_merkle_tree_v2(height=12000) to catch up missing accounts and
241    // fails because no tree exists in storage at that height.
242    let is_epoch_boundary = version >= EPOCH_REWARD_VERSION
243        && is_last_block(
244            block_number,
245            instance
246                .epoch_height
247                .expect("epoch_height should be set for version > V3"),
248        );
249
250    if has_changed_accounts || is_epoch_boundary {
251        storage
252            .save_and_gc_reward_tree_v2(
253                instance,
254                block_number,
255                version,
256                &state.reward_merkle_tree_v2,
257            )
258            .await
259            .context("failed to save and gc reward merkle tree v2")?;
260    }
261
262    storage
263        .persist_reward_proofs(instance, block_number, version)
264        .await
265        .context("failed to persist reward proofs")?;
266
267    // The nodes and the new head commit together, so no node version ever exists above the head:
268    // the head snapshot is the newest version of everything it touches, which is what keeps it
269    // intact under pruning and lets the loop resume from it after a crash.
270    tracing::debug!("storing state update");
271    let mut tx = storage
272        .write()
273        .await
274        .context("opening transaction for state update")?;
275
276    store_state_update(&mut tx, block_number, version, &state, &delta).await?;
277
278    if parent_chain_config != state.chain_config {
279        let cf = state
280            .chain_config
281            .resolve()
282            .context("failed to resolve to chain config")?;
283
284        tx.insert_chain_config(cf).await?;
285    }
286
287    tracing::debug!(block_number, "updating state height");
288    UpdateStateData::<SeqTypes, _, { BlockMerkleTree::ARITY }>::set_last_state_height(
289        &mut tx,
290        block_number as usize,
291    )
292    .await
293    .context("setting state height")?;
294
295    tx.commit().await?;
296
297    Ok(state)
298}

300async fn store_genesis_state<S>(
301    storage: &S,
302    chain_config: ChainConfig,
303    state: &ValidatedState,
304) -> anyhow::Result<()>
305where
306    S: SequencerStateDataSource,
307    for<'a> S::Transaction<'a>: SequencerStateUpdate,
308{
309    ensure!(
310        state.block_merkle_tree.num_leaves() == 0,
311        "genesis state with non-empty block tree is unsupported"
312    );
313
314    let mut tx = storage
315        .write()
316        .await
317        .context("starting transaction for genesis state")?;
318
319    // Insert fee merkle tree nodes
320    for (account, _) in state.fee_merkle_tree.iter() {
321        let proof = match state.fee_merkle_tree.universal_lookup(account) {
322            LookupResult::Ok(_, proof) => proof,
323            LookupResult::NotFound(proof) => proof,
324            LookupResult::NotInMemory => bail!("missing merkle path for fee account {account}"),
325        };
326        let path: Vec<usize> =
327            <FeeAccount as ToTraversalPath<{ FeeMerkleTree::ARITY }>>::to_traversal_path(
328                account,
329                state.fee_merkle_tree.height(),
330            );
331
332        UpdateStateData::<SeqTypes, FeeMerkleTree, { FeeMerkleTree::ARITY }>::insert_merkle_nodes(
333            &mut tx, proof, path, 0,
334        )
335        .await
336        .context("failed to store fee merkle nodes")?;
337    }
338
339    tx.insert_chain_config(chain_config).await?;
340
341    tx.commit().await?;
342
343    // Store the genesis reward tree at height 0 so catchup can find it.
344    let tree_data: RewardMerkleTreeV2Data = (&state.reward_merkle_tree_v2)
345        .try_into()
346        .context("serializing genesis reward tree")?;
347    let tree_bytes = bincode::serialize(&tree_data).context("serializing genesis reward tree")?;
348    storage
349        .persist_tree(0, tree_bytes)
350        .await
351        .context("storing genesis reward merkle tree")?;
352
353    Ok(())
354}

356/// How long the loop waits on the local leaf stream before checking whether the data pruner has
357/// passed the height it is waiting for.
358const PRUNED_LEAF_CHECK_INTERVAL: Duration = Duration::from_secs(10);
359
360#[tracing::instrument(skip_all)]
361pub(crate) async fn update_state_storage_loop<T>(
362    storage: Arc<T>,
363    instance: impl Future<Output = NodeState>,
364    provider: Arc<LightClientProvider>,
365) -> anyhow::Result<()>
366where
367    T: SequencerStateDataSource,
368    for<'a> T::Transaction<'a>: SequencerStateUpdate,
369{
370    let mut instance = instance.await;
371    // Use a separate rewards calculator for the state loop so it doesn't
372    // interfere with consensus, which may be on a very different epoch.
373    instance.epoch_rewards_calculator = Arc::new(Mutex::new(EpochRewardsCalculator::new()));
374
375    // Resume from the newest snapshot in storage. The loop cannot start anywhere else:
376    // `from_header` gives it a bare state, and its catchup fills in the block frontier and any fee
377    // account it has not seen from the parent's snapshot in this database, which exists only up to
378    // the head. The state pruner in turn never advances to the head, so that snapshot stays
379    // readable however far behind the chain the loop has fallen.
380    let last_height = storage.get_last_state_height().await?;
381    let current_height = storage.block_height().await?;
382    tracing::info!(
383        node_id = instance.node_id,
384        last_height,
385        current_height,
386        "updating state storage"
387    );
388
389    let parent_leaf = leaf_at(&storage, &provider, last_height as u64).await;
390    let mut parent_state = ValidatedState::from_header(parent_leaf.block_header());
391
392    // Seed the parent's reward tree from storage.
393    //
394    // `from_header` starts the reward tree empty, and the epoch boundary only
395    // repopulates it when the previous epoch actually distributed rewards. If the
396    // previous epoch predates rewards (e.g. the first boundary after a V4→V5
397    // upgrade, where the previous epoch is pre-V5), `handle_epoch_rewards` returns
398    // zero and passes the empty tree through unchanged. So after a restart in such
399    // an epoch the tree stays empty until the next boundary, where the save in
400    // `update_state_storage` fails as it serializes the tree as full key-value pairs
401    // and bails because the accounts are missing.
402    //
403    // Load the parent's tree from storage to avoid this. Doing it once is enough
404    // every later iteration carries the tree forward in `parent_state`.
405    if parent_leaf.block_header().version() > EPOCH_VERSION && parent_leaf.height() > 0 {
406        // The tree is only written at epoch boundaries, so the parent height
407        // usually has no row; fall back to the most recent tree at or below it.
408        let reward_merkle_tree_v2 = match storage
409            .load_reward_merkle_tree_v2(parent_leaf.height())
410            .await
411        {
412            Ok(tree) => tree,
413            Err(_) => storage
414                .load_latest_reward_merkle_tree_v2(parent_leaf.height())
415                .await
416                .context(
417                    "Error starting the state storage update loop: failed to load \
418                     RewardMerkleTreeV2 for the previous height",
419                )?,
420        };
421
422        // The fallback returns the latest tree at or below the parent height,
423        // which is the parent's tree only if their roots match (they do within an
424        // epoch). Verify against the root the parent header commits to.
425        let parent_reward_root = parent_leaf
426            .block_header()
427            .reward_merkle_tree_root()
428            .right()
429            .context("V5+ parent header must have a v2 reward root")?;
430        ensure!(
431            reward_merkle_tree_v2.tree.commitment() == parent_reward_root,
432            "loaded reward tree root {} does not match parent header {parent_reward_root}",
433            reward_merkle_tree_v2.tree.commitment(),
434        );
435
436        parent_state.reward_merkle_tree_v2 = reward_merkle_tree_v2.tree;
437    }
438
439    if last_height == 0 {
440        // If the last height is 0, we need to insert the genesis state, since this state is
441        // never the result of a state update and thus is not inserted in the loop below.
442        tracing::info!("storing genesis merklized state");
443        store_genesis_state(&*storage, instance.chain_config, &instance.genesis_state)
444            .await
445            .context("storing genesis state")?;
446    }
447
448    follow_leaves(&storage, &instance, &provider, parent_leaf, parent_state).await;
449    Ok(())
450}

452/// Apply every leaf above `parent_leaf` as it arrives. Leaves come from the local stream, except
453/// those the data pruner has already deleted: the availability layer never fetches a leaf at or
454/// below the data pruned height, so the stream would wait for one forever. Those come from peers,
455/// in runs. Returns when the stream ends, or when a leaf from peers does not extend the parent.
456async fn follow_leaves<T>(
457    storage: &Arc<T>,
458    instance: &NodeState,
459    provider: &LightClientProvider,
460    mut parent_leaf: Leaf2,
461    mut parent_state: ValidatedState,
462) where
463    T: SequencerStateDataSource,
464    for<'a> T::Transaction<'a>: SequencerStateUpdate,
465{
466    // Peers serve at most this many leaves per range request.
467    let limit = availability::Options::default().small_object_range_limit as u64;
468    let mut next_height = parent_leaf.height() + 1;
469    loop {
470        while let Some(pruned) = pruned_past(&**storage, next_height).await {
471            let end = (next_height + limit).min(pruned + 1);
472            tracing::info!(
473                next_height,
474                end,
475                pruned,
476                "fetching leaves below the data pruned height from peers"
477            );
478            for leaf in fetch_pruned_leaves(provider, next_height..end).await {
479                // The leaves are verified, so they are the decided leaves at these heights, and a
480                // refetch returns them again. A mismatch means the loop's parent is wrong, which
481                // no retry fixes.
482                if leaf.parent_commitment() != parent_leaf.commit() {
483                    tracing::error!(
484                        height = next_height,
485                        parent = %parent_leaf.commit(),
486                        leaf_parent = %leaf.parent_commitment(),
487                        "fetched leaf does not extend the state loop's parent; stopping the loop"
488                    );
489                    return;
490                }
491                apply_leaf(storage, instance, &mut parent_leaf, &mut parent_state, leaf).await;
492                next_height += 1;
493            }
494        }
495
496        let mut leaves = storage.subscribe_leaves(next_height as usize).await;
497        loop {
498            match timeout(PRUNED_LEAF_CHECK_INTERVAL, leaves.next()).await {
499                Ok(Some(leaf)) => {
500                    next_height = leaf.height() + 1;
501                    apply_leaf(
502                        storage,
503                        instance,
504                        &mut parent_leaf,
505                        &mut parent_state,
506                        leaf.leaf().clone(),
507                    )
508                    .await;
509                },
510                Ok(None) => return,
511                // Still waiting. If the data pruner has passed this height in the meantime, the
512                // stream will never deliver it.
513                Err(_) => {
514                    if pruned_past(&**storage, next_height).await.is_some() {
515                        break;
516                    }
517                },
518            }
519        }
520    }
521}

523/// The data pruned height when it is at or above `height`. The leaf and header there are out of
524/// local reach: the pruner deleted them, and the availability layer never fetches below its
525/// marker. A failed read counts as not pruned; every caller asks again before long, and the loop
526/// must not end over a transient database error.
527async fn pruned_past<T>(storage: &T, height: u64) -> Option<u64>
528where
529    T: SequencerStateDataSource,
530{
531    match storage.load_pruned_height().await {
532        Ok(pruned) => pruned.filter(|pruned| height <= *pruned),
533        Err(err) => {
534            tracing::warn!(height, "failed to load the data pruned height: {err:#}");
535            None
536        },
537    }
538}

540/// The leaf at `height`, from local storage, or from peers once the data pruner has deleted it.
541async fn leaf_at<T>(storage: &Arc<T>, provider: &LightClientProvider, height: u64) -> Leaf2
542where
543    T: SequencerStateDataSource,
544{
545    loop {
546        if let Some(pruned) = pruned_past(&**storage, height).await {
547            tracing::warn!(
548                height,
549                pruned,
550                "leaf is at or below the data pruned height; fetching it from peers"
551            );
552            return fetch_pruned_leaves(provider, height..height + 1)
553                .await
554                .pop()
555                .expect("a one-height range yields exactly one leaf");
556        }
557        let local = AvailabilityDataSource::get_leaf(&**storage, height as usize).await;
558        // On a timeout the data pruner may have passed this height in the meantime, so check
559        // again before waiting further.
560        if let Ok(leaf) = timeout(PRUNED_LEAF_CHECK_INTERVAL, local).await {
561            return leaf.leaf().clone();
562        }
563        tracing::info!(
564            height,
565            "leaf is not in local storage yet; still waiting for it"
566        );
567    }
568}

570/// The leaves at `heights`, which the data pruned height says are gone, from the query module's
571/// peers through its light client, retrying until they arrive. The light client verifies leaves
572/// of every protocol version, and one proof covers the run: it proves the last leaf final and
573/// chains the rest to it. The leaves are not stored in this database, since the loop needs them
574/// once and the pruner would only delete them again.
575async fn fetch_pruned_leaves(provider: &LightClientProvider, heights: Range<u64>) -> Vec<Leaf2> {
576    let start = heights.start as usize;
577    let end = heights.end as usize;
578    let mut delay = Duration::from_secs(1);
579    loop {
580        if let Some(light_client) = provider.light_client().await {
581            match light_client.fetch_leaves_in_range(start, end).await {
582                Ok(leaves) => return leaves.into_iter().map(|leaf| leaf.leaf().clone()).collect(),
583                Err(err) => tracing::warn!(
584                    ?heights,
585                    ?delay,
586                    err = %format_args!("{err:#}"),
587                    "fetching leaves failed, retrying"
588                ),
589            }
590        }
591        sleep(delay).await;
592        delay = (delay * 2).min(Duration::from_secs(60));
593    }
594}

596/// Apply `leaf` to the merklized state, retrying until it is stored, then make it the parent.
597async fn apply_leaf<T>(
598    storage: &Arc<T>,
599    instance: &NodeState,
600    parent_leaf: &mut Leaf2,
601    parent_state: &mut ValidatedState,
602    leaf: Leaf2,
603) where
604    T: SequencerStateDataSource,
605    for<'a> T::Transaction<'a>: SequencerStateUpdate,
606{
607    // The parent snapshot is the state head, which this database always keeps, so read it from
608    // here only: peers may have pruned it. When the data pruner has deleted the parent's leaf and
609    // header, the catchup reads the snapshot by the roots in the parent state. Reward accounts do
610    // not use this catchup; `apply_header` reads them through the node's catchup. Retries are off
611    // because the loop below retries.
612    let catchup = SqlStateCatchup::new(storage.clone(), BackoffParams::disabled());
613    loop {
614        tracing::debug!(
615            height = leaf.height(),
616            node_id = instance.node_id,
617            ?leaf,
618            "updating persistent merklized state"
619        );
620        match update_state_storage(
621            parent_state,
622            storage,
623            instance,
624            &catchup,
625            parent_leaf,
626            &leaf,
627        )
628        .await
629        {
630            Ok(state) => {
631                *parent_leaf = leaf;
632                *parent_state = state;
633                return;
634            },
635            Err(err) => {
636                tracing::error!(height = leaf.height(), "failed to update state: {err:#}");
637                // If we fail, delay for a second and retry.
638                sleep(Duration::from_secs(1)).await;
639            },
640        }
641    }
642}
